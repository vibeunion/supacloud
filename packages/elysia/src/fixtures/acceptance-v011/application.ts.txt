import { Body, Controller, Get, Injectable, Module, Post, Query } from "@supacloud/app";
import { type Static, type StaticDecode } from "typebox";
import { status } from "elysia";
import { BodySchema, ConflictSchema, QuerySchema, ResultSchema } from "./contracts";

@Injectable()
@Controller("/acceptance")
export class AcceptanceController {
  @Get("/item", { query: QuerySchema, response: ResultSchema })
  read(@Query("count") count: StaticDecode<typeof QuerySchema>["count"]): Static<typeof ResultSchema> {
    return { note: null, count };
  }

  @Post("/item", { body: BodySchema, responses: { 200: ResultSchema, 409: ConflictSchema } })
  write(@Body() body: Static<typeof BodySchema>) {
    return body.note === "conflict"
      ? status(409, { conflict: true as const })
      : { note: body.note ?? null, count: 0 };
  }
}

@Module({ name: "acceptance", controllers: [AcceptanceController] })
export class AcceptanceModule {}
